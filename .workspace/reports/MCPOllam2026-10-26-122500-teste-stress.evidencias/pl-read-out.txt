# /etc/systemd/system/nvidia-powercap.service
[Unit]
Description=NVIDIA GPU power cap for stable 4-GPU inference (240W MAX post-riser-fix)
After=multi-user.target nvidia-persistenced.service
Wants=nvidia-persistenced.service

[Service]
Type=oneshot
RemainAfterExit=yes
# Conservador (owner 2026-10-23): pos Xid 79 'GPU fell off the bus' na placa PCI 01:00 sob carga a 240W.
# Power limit 170W nas 4 GPUs; SEM locked clocks (deixa o clock gerir-se = menor stress/pico).
ExecStart=/bin/bash -c 'for i in $(nvidia-smi --query-gpu=index --format=csv,noheader); do nvidia-smi -i "$i" -pm 1; nvidia-smi -i "$i" -pl 170; nvidia-smi -i "$i" -rgc; done'
Restart=on-failure
RestartSec=15

[Install]
WantedBy=multi-user.target
